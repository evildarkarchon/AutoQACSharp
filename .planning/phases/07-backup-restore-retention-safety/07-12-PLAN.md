```csharp
public ProgressWindow()
{
    InitializeComponent();
    DataContextChanged += OnDataContextChanged;
}

private void OnDataContextChanged(object? sender, EventArgs e)
{
    if (_subscribedViewModel != null)
    {
        _subscribedViewModel.CloseRequested -= OnCloseRequested;
        _subscribedViewModel = null;
    }
    if (DataContext is ProgressViewModel viewModel && !_disposeHandled)
    {
        _subscribedViewModel = viewModel;
        viewModel.CloseRequested += OnCloseRequested;
    }
}

private void OnCloseRequested(object? sender, EventArgs e)
{
    DisposeViewModelIfNeeded();
    Close();
}

protected override void OnClosed(EventArgs e)
{
    DisposeViewModelIfNeeded();
    base.OnClosed(e);
}

private void DisposeViewModelIfNeeded()
{
    if (_disposeHandled) return;
    _disposeHandled = true;
    DataContextChanged -= OnDataContextChanged;
    if (_subscribedViewModel != null)
    {
        _subscribedViewModel.CloseRequested -= OnCloseRequested;
        _subscribedViewModel.Dispose();
        _subscribedViewModel = null;
    }
}
```

**Reviewer consensus on dual subscription:** Gemini reads the new wiring as defense in depth; the agent and Codex flagged the redundancy risk. Consensus resolution adopted by this plan: **add the wiring AND a one-line `// Defense in depth:` code comment in `ShowProgressAsync` explaining why both subscription paths exist**, plus keep the local `progressDisposed` closure guard idempotent (so even if `ProgressWindow.OnClosed → DisposeViewModelIfNeeded` already disposed the VM, the local function short-circuits safely). This matches Avalonia's idempotent `Window.Close()` semantics — a second `Close()` is a no-op.

**Reviewer consensus on test brittleness:** Gemini, the agent, and Codex all flagged the previous five-substring assertion as brittle. Resolution: replace exact substring matches with regex tolerant of method-group syntax (`Closed += OnProgressClosed;`) and alternate guard variable names (`disposed`, `cleaned`, `done`, `progressDisposed`).
