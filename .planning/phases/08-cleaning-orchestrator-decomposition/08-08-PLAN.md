```csharp
public async Task<PluginCleaningResult> FinalizeAsync(
    PluginInfo plugin,
    GameType gameType,
    string xEditDir,
    PluginRunnerOutput runnerOutput,
    TerminationFinalizeContext terminationContext,
    CancellationToken ct)
{
    var result = runnerOutput.LastAttemptResult;        // CleaningResult { Success, Status, Message, TimedOut, ... }
    CleaningStatistics? logStats = null;
    string? logParseWarning = null;
    var finalStatus = result.Status;

    if (!terminationContext.ProcessMayStillBeRunning && !terminationContext.StopWasRequested
        && result.Status != CleaningStatus.Skipped)
    {
        var logResult = await logFileService.ReadLogContentAsync(...).ConfigureAwait(false);
        if (logResult.Warning != null) { logParseWarning = logResult.Warning; }

        if (logResult.LogLines.Count > 0)
        {
            logStats = outputParser.ParseOutput(logResult.LogLines);

            // CURRENT BUGGY GATE — promotes ANY non-skipped result with completion line + zero stats:
            var hasCompletionLine = logResult.LogLines.Any(outputParser.IsCompletionLine);
            if (hasCompletionLine && logStats is { ItemsRemoved: 0, ItemsUndeleted: 0, ItemsSkipped: 0, PartialFormsCreated: 0 })
            {
                finalStatus = CleaningStatus.AlreadyClean;
            }
        }

        if (logResult.ExceptionContent != null)
        {
            logParseWarning = logResult.ExceptionContent;
            finalStatus = CleaningStatus.Failed;
        }
    }
    else if (terminationContext.StopWasRequested || terminationContext.ProcessMayStillBeRunning)
    {
        logParseWarning = "xEdit was terminated -- no log available";
    }

    return new PluginCleaningResult
    {
        PluginName = plugin.FileName,
        Status = finalStatus,
        Success = result.Success,                    // BUG WR-01: ignores finalStatus override
        Message = ...,
        Duration = runnerOutput.Duration,
        Statistics = logStats,
        LogParseWarning = logParseWarning
    };
}
```
