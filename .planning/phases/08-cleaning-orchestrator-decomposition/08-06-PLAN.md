    ```csharp
    [Fact]
    public void Cleaning_Source_NoFileParallelizesPluginLoop()
    {
        // INV-8.2: Sequential xEdit cleaning is a hard runtime requirement (CLAUDE.md, ROADMAP.md
        // out-of-scope, Phase 8 D-09/D-10 locks). Source-level guard prevents accidental parallelization
        // anywhere in the cleaning service tier.
        //
        // Locate the AutoQAC project's Services/Cleaning folder relative to test assembly location.
        var cleaningDir = LocateCleaningSourceFolder();

        var filesToScan = new[]
        {
            "CleaningOrchestrator.cs",
            "CleaningPreflight.cs",
            "BackupSessionCoordinator.cs",
            "PluginCleaningRunner.cs",
            "PluginResultFinalizer.cs",
            "CleaningTerminationCoordinator.cs"
        };

        var prohibitedTokens = new[]
        {
            "Parallel.ForEach",
            "Parallel.ForEachAsync",
            "Task.WhenAll(",
            "Task.WhenAny(",
            // Task.Run is allowed for fire-and-forget infrastructure work; we only flag it inside the foreach plugin loop.
            // The simpler structural guard checks for `Parallel` and `WhenAll` which are unambiguous.
        };

        foreach (var file in filesToScan)
        {
            var path = Path.Combine(cleaningDir, file);
            File.Exists(path).Should().BeTrue($"Phase 8 file {file} must exist under AutoQAC/Services/Cleaning/");
            var content = File.ReadAllText(path);

            foreach (var token in prohibitedTokens)
            {
                content.Should().NotContain(token,
                    $"{file} must not parallelize cleaning work — sequential xEdit is a hard runtime invariant (Phase 8 D-10).");
            }
        }
    }

    private static string LocateCleaningSourceFolder()
    {
        // Walk up from test assembly location to repo root, then into AutoQAC/Services/Cleaning.
        var dir = new DirectoryInfo(AppContext.BaseDirectory);
        while (dir is not null && !File.Exists(Path.Combine(dir.FullName, "AutoQACSharp.slnx")))
        {
            dir = dir.Parent;
        }
        if (dir is null) throw new InvalidOperationException("Could not locate solution root from test assembly base directory.");
        return Path.Combine(dir.FullName, "AutoQAC", "Services", "Cleaning");
    }
    ```
